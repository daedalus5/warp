# tile_empty Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a new Warp tile primitive `wp.tile_empty(shape, dtype, storage)` that allocates a tile of uninitialized data — the `np.empty` analog of `wp.tile_zeros` — backed by a refactor that makes register tiles follow the same allocate-then-assign pattern shared tiles already use.

**Architecture:** `tile_register_t`'s default constructor becomes a no-op (mirroring shared's `tile_alloc_empty`). A new `operator=(const T&)` on register tiles handles scalar broadcast, so `tile_zeros` continues to work via single-loop assignment. A `tile_no_init_t` tag type, returned by `tile_empty<...>()`, flows through a no-op `operator=` overload on both tile types, leaving the destination untouched.

**Tech Stack:** C++/CUDA (Warp native), Python (Warp builtins, value/dispatch funcs), unittest, NumPy.

**Spec:** See `design/tile-empty.md` for motivation, contract, and design rationale.

def tile_empty_value_func(arg_types: Mapping[str, type], arg_values: Mapping[str, Any]):
    # return generic type (for doc builds)
    if arg_types is None:
        return tile(dtype=Any, shape=tuple[int, ...])

    shape = extract_tuple(arg_values["shape"], as_constant=True)

    if None in shape:
        raise ValueError("Tile functions require shape to be a compile time constant.")

    if "dtype" not in arg_values:
        raise TypeError("tile_empty() missing required keyword argument 'dtype'")

    if "storage" not in arg_values:
        raise TypeError("tile_empty() missing required keyword argument 'storage'")

    if arg_values["storage"] not in {"shared", "register"}:
        raise ValueError(f"Invalid value for 'storage': {arg_values['storage']!r}. Expected 'shared' or 'register'.")

    dtype = arg_values["dtype"]

    return tile(dtype=dtype, shape=shape, storage=arg_values["storage"])


def tile_empty_dispatch_func(arg_types: Mapping[str, type], return_type: Any, arg_values: Mapping[str, Var]):
    shape = extract_tuple(arg_values["shape"], as_constant=True)

    if None in shape:
        raise ValueError("Tile functions require shape to be a compile time constant.")

    dtype = arg_values["dtype"]

    template_args = []
    template_args.append(dtype)
    template_args.extend(shape)

    return ([], template_args)


add_builtin(
    "tile_empty",
    input_types={"shape": tuple[int, ...], "dtype": Any, "storage": str},
    defaults={"storage": "register", "dtype": float},
    value_func=tile_empty_value_func,
    dispatch_func=tile_empty_dispatch_func,
    variadic=False,
    is_differentiable=False,
    doc="""Allocate a tile of uninitialized items.

    The tile's contents are undefined; the caller is responsible for overwriting
    every element before any read. This matches the semantics of ``numpy.empty``.

    For accumulator patterns (``a += ...``), use :func:`tile_zeros` instead -
    accumulation reads the prior value and would propagate uninitialized data.
    Use ``tile_empty`` only when the first operation after construction is a
    full overwrite (a ``tile_load``, a tile-typed assignment, or a complete
    element-wise fill).

    Args:
        shape: Shape of the output tile
        dtype: Data type of output tile's elements (default float)
        storage: The storage location for the tile: ``"register"`` for registers
            (default) or ``"shared"`` for shared memory.

    Returns:
        An uninitialized tile with the requested shape and data type.""",
    group="Tile Primitives",
    export=False,

## Task 11: Scratch benchmark and verify the perf claim (R4)

**Files:**

- [ ] **Step 11.1: Write the benchmark script**

```python
"""Scratch benchmark: tile_empty vs tile_zeros.

R4 (from design/tile-empty.md): tile_empty must be at least as fast as
tile_zeros across {register, shared} x {float, vec3} x several shapes.
Asserts a soft bound of tile_empty <= tile_zeros * 1.05 (5% noise slack).
"""

import numpy as np

import warp as wp

ITERS = 200
WARMUP = 50
BLOCK_DIM = 128


def make_zeros_kernel(M, N, dtype, storage):
    @wp.kernel
    def k(out: wp.array2d(dtype=dtype)):
        for _ in range(20):
            t = wp.tile_zeros(shape=(M, N), dtype=dtype, storage=storage)
            wp.tile_store(out, t)
    return k


def make_empty_kernel(M, N, dtype, storage):
    # The fair comparison: empty + a tile_load that overwrites.
    # The 'load' is common to both kernels in the comparison; it cancels.
    @wp.kernel
    def k(src: wp.array2d(dtype=dtype), out: wp.array2d(dtype=dtype)):
        for _ in range(20):
            t = wp.tile_empty(shape=(M, N), dtype=dtype, storage=storage)
            t = wp.tile_load(src, shape=(M, N))
            wp.tile_store(out, t)
    return k


def make_zeros_then_load_kernel(M, N, dtype, storage):
    @wp.kernel
    def k(src: wp.array2d(dtype=dtype), out: wp.array2d(dtype=dtype)):
        for _ in range(20):
            t = wp.tile_zeros(shape=(M, N), dtype=dtype, storage=storage)
            t = wp.tile_load(src, shape=(M, N))
            wp.tile_store(out, t)
    return k


def time_kernel(kernel, args, device, iters=ITERS, warmup=WARMUP):
    for _ in range(warmup):
        wp.launch_tiled(kernel, dim=[1], inputs=args, block_dim=BLOCK_DIM, device=device)
    wp.synchronize_device(device)

    start = wp.Event(enable_timing=True)
    end = wp.Event(enable_timing=True)
    wp.record_event(start)
    for _ in range(iters):
        wp.launch_tiled(kernel, dim=[1], inputs=args, block_dim=BLOCK_DIM, device=device)
    wp.record_event(end)
    return wp.get_event_elapsed_time(start, end) / iters * 1e3  # microseconds per launch


def main():
    device = wp.get_cuda_device(0)

    configs = [
        # (M, N, dtype, storage)
        (16,  16,  wp.float32, "register"),
        (64,  64,  wp.float32, "register"),
        (16,  16,  wp.float32, "shared"),
        (64,  64,  wp.float32, "shared"),
        (128, 128, wp.float32, "shared"),
        (16,  16,  wp.vec3,    "register"),
        (16,  16,  wp.vec3,    "shared"),
    ]

    print(f"{'shape':>10} {'dtype':>10} {'storage':>10} {'tile_zeros (us)':>18} {'tile_empty (us)':>18} {'ratio':>8}")
    print("-" * 80)

    failures = []
    for M, N, dtype, storage in configs:
        src = wp.zeros((M, N), dtype=dtype, device=device)
        out = wp.zeros((M, N), dtype=dtype, device=device)

        zeros_k = make_zeros_then_load_kernel(M, N, dtype, storage)
        empty_k = make_empty_kernel(M, N, dtype, storage)

        t_zeros = time_kernel(zeros_k, [src, out], device)
        t_empty = time_kernel(empty_k, [src, out], device)
        ratio = t_empty / t_zeros if t_zeros > 0 else float('inf')

        dtype_name = dtype.__name__ if hasattr(dtype, "__name__") else str(dtype)
        print(f"{f'{M}x{N}':>10} {dtype_name:>10} {storage:>10} {t_zeros:>18.3f} {t_empty:>18.3f} {ratio:>8.2f}")

        if ratio > 1.05:
            failures.append((M, N, dtype_name, storage, ratio))

    print()
    if failures:
        print("FAIL: tile_empty exceeded tile_zeros * 1.05 in:")
        for f in failures:
            print(f"  {f[0]}x{f[1]} {f[2]} {f[3]}: ratio {f[4]:.2f}")
        raise SystemExit(1)
    else:
        print("PASS: tile_empty <= tile_zeros * 1.05 across all configs")


if __name__ == "__main__":
    main()
```
